# NetHack SDL3 fork -- standalone build:
#
#   cmake -S sys/sdl3 -B build
#   cmake --build build
#
# Only a C/C++ compiler and CMake are required; SDL3, Lua, and
# PDCursesMod are fetched over HTTPS and verified by SHA256.

cmake_minimum_required(VERSION 3.28)
project(nethack-sdl3 C CXX)

get_filename_component(NH_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)

if(CMAKE_CROSSCOMPILING AND NOT CMAKE_CROSSCOMPILING_EMULATOR)
    message(FATAL_ERROR "This build runs its own host tools (makedefs, "
        "dlb, bin2c, hex2c) at build time; build natively or set "
        "CMAKE_CROSSCOMPILING_EMULATOR.")
endif()

# --- options ----------------------------------------------------------
# Curses backend: "sdl3" is the real UI; "wincon"/"vt" use PDCursesMod's
# own console backends (bring-up and debugging).
set(NETHACK_UI "sdl3" CACHE STRING "curses backend: sdl3, wincon, or vt")
if(WIN32)
    option(NETHACK_SDL3_PORTABLE "store all runtime files next to the exe" ON)
else()
    option(NETHACK_SDL3_PORTABLE "store all runtime files next to the exe" OFF)
endif()
option(NETHACK_SDL3_CONSOLE "build as a console-subsystem app (Windows)" OFF)

# GCC 15+ defaults to C23, which neither this codebase nor PDCursesMod
# is ready for (PDCursesMod issue #333 requires gnu17).
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_EXTENSIONS ON)
set(CMAKE_CXX_STANDARD 17)

if(WIN32)
    add_compile_options(-mms-bitfields)
endif()

set(GEN "${CMAKE_BINARY_DIR}/gen")
set(GENINC "${CMAKE_BINARY_DIR}/geninc")
set(STAGE "${CMAKE_BINARY_DIR}/stage")
file(MAKE_DIRECTORY "${GEN}" "${GENINC}" "${STAGE}/dat" "${STAGE}/run")

# --- third-party sources (FetchContent, URL + SHA256 only) ------------
include(FetchContent)

# Hash matches the repo's own submodules/CHKSUMS entry.
FetchContent_Declare(lua
    URL https://www.lua.org/ftp/lua-5.4.8.tar.gz
    URL_HASH SHA256=4f18ddae154e793e46eeab727c59ef1c0c0c2b744e7b94219710d76f530629ae
    SOURCE_SUBDIR do-not-add-subdirectory)

FetchContent_Declare(pdcursesmod
    URL https://github.com/Bill-Gray/PDCursesMod/archive/refs/tags/v4.5.4.tar.gz
    URL_HASH SHA256=d5efc7f2b7107abe382bdf8bac0a9bfd8e716facbca2bb9cf12dfeb8e1122c4b
    SOURCE_SUBDIR do-not-add-subdirectory)

set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
FetchContent_Declare(SDL3
    URL https://github.com/libsdl-org/SDL/releases/download/release-3.4.14/SDL3-3.4.14.tar.gz
    URL_HASH SHA256=30d4aa2b3037718142b32dffd4e72f917ebb6cc5227150e7bb9c45efb2153aeb)

FetchContent_MakeAvailable(lua pdcursesmod SDL3)

set(LUASRC "${lua_SOURCE_DIR}/src")

# --- generated header: nhlua.h ----------------------------------------
# The one generated header in the whole build (hack.h includes it).
# The lua_error redeclaration attaches NORETURN, as the stock makefiles do.
file(WRITE "${GENINC}/nhlua.h"
"/* nhlua.h - generated by CMake */
#include \"lua.h\"
ATTRNORETURN LUA_API int (lua_error) (lua_State *L) NORETURN;
#include \"lualib.h\"
#include \"lauxlib.h\"
/*nhlua.h*/
")

# --- Lua --------------------------------------------------------------
file(GLOB LUA_SOURCES "${LUASRC}/*.c")
list(FILTER LUA_SOURCES EXCLUDE REGEX "(lua|luac)\\.c$") # lua.c luac.c onelua.c
add_library(nh_lua STATIC ${LUA_SOURCES})
target_include_directories(nh_lua PUBLIC "${LUASRC}")
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_compile_definitions(nh_lua PRIVATE LUA_USE_LINUX)
    target_link_libraries(nh_lua PUBLIC m ${CMAKE_DL_LIBS})
endif()

# --- host tools -------------------------------------------------------
set(NH_HOST_DEFS NETHACK_SDL3 DLB)

add_executable(nh_makedefs
    "${NH_ROOT}/util/makedefs.c"   # #includes src/mdlib.c
    "${NH_ROOT}/src/monst.c"
    "${NH_ROOT}/src/objects.c"
    "${NH_ROOT}/src/date.c"
    "${NH_ROOT}/src/alloc.c"
    "${NH_ROOT}/src/hacklib.c"
    "${NH_ROOT}/util/panic.c")

add_executable(nh_dlb
    "${NH_ROOT}/util/dlb_main.c"
    "${NH_ROOT}/src/dlb.c"
    "${NH_ROOT}/src/alloc.c"
    "${NH_ROOT}/src/hacklib.c"
    "${NH_ROOT}/util/panic.c")

foreach(tool nh_makedefs nh_dlb)
    target_compile_definitions(${tool} PRIVATE ${NH_HOST_DEFS})
    target_include_directories(${tool} PRIVATE
        "${GENINC}" "${NH_ROOT}/include" "${LUASRC}")
endforeach()

add_executable(nh_bin2c "${CMAKE_CURRENT_SOURCE_DIR}/tools/bin2c.c")
add_executable(nh_hex2c "${CMAKE_CURRENT_SOURCE_DIR}/tools/hex2c.c")

# --- data pipeline ----------------------------------------------------
# makedefs hardcodes ../dat/ relative to its CWD (and drops temp files
# there), so it runs from a staged tree inside the build directory.
set(MD_INPUTS data.base rumors.tru rumors.fal oracles.txt
              engrave.txt epitaph.txt bogusmon.txt)
set(MD_OUTPUTS data rumors oracles engrave epitaph bogusmon)

set(MD_INPUT_PATHS)
foreach(f IN LISTS MD_INPUTS)
    list(APPEND MD_INPUT_PATHS "${NH_ROOT}/dat/${f}")
endforeach()
set(MD_OUTPUT_PATHS)
foreach(f IN LISTS MD_OUTPUTS)
    list(APPEND MD_OUTPUT_PATHS "${STAGE}/dat/${f}")
endforeach()

add_custom_command(
    OUTPUT ${MD_OUTPUT_PATHS}
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
            ${MD_INPUT_PATHS} "${STAGE}/dat"
    COMMAND "$<TARGET_FILE:nh_makedefs>" -drhs
    WORKING_DIRECTORY "${STAGE}/run"
    DEPENDS nh_makedefs ${MD_INPUT_PATHS}
    COMMENT "makedefs: generating data files")

file(GLOB NH_LUA_DATA CONFIGURE_DEPENDS "${NH_ROOT}/dat/*.lua")
set(NH_DLB_FIXED help hh cmdhelp keyhelp history opthelp optmenu usagehlp
                 wizhelp luahelper license tribute)
set(NH_DLB_FIXED_PATHS)
foreach(f IN LISTS NH_DLB_FIXED)
    list(APPEND NH_DLB_FIXED_PATHS "${NH_ROOT}/dat/${f}")
endforeach()
add_custom_command(
    OUTPUT "${STAGE}/dat/nhdat"
    COMMAND ${CMAKE_COMMAND}
            -DSRCDAT=${NH_ROOT}/dat
            -DSTAGEDAT=${STAGE}/dat
            -DDLB_EXE=$<TARGET_FILE:nh_dlb>
            "-DFIXED=${NH_DLB_FIXED}"
            -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/pack_dlb.cmake"
    DEPENDS nh_dlb ${MD_OUTPUT_PATHS} ${NH_LUA_DATA} ${NH_DLB_FIXED_PATHS}
            "${CMAKE_CURRENT_SOURCE_DIR}/cmake/pack_dlb.cmake"
    COMMENT "dlb: packing nhdat"
    VERBATIM)

# --- embedded blobs and font ------------------------------------------
add_custom_command(
    OUTPUT "${GEN}/nhdat_blob.c"
    COMMAND "$<TARGET_FILE:nh_bin2c>" "${STAGE}/dat/nhdat"
            "${GEN}/nhdat_blob.c" nhdat_blob
    DEPENDS nh_bin2c "${STAGE}/dat/nhdat"
    COMMENT "bin2c: embedding nhdat")

add_custom_command(
    OUTPUT "${GEN}/nh_symbols_blob.c"
    COMMAND "$<TARGET_FILE:nh_bin2c>" "${NH_ROOT}/dat/symbols"
            "${GEN}/nh_symbols_blob.c" nh_symbols_blob
    DEPENDS nh_bin2c "${NH_ROOT}/dat/symbols"
    COMMENT "bin2c: embedding symbols")

add_custom_command(
    OUTPUT "${GEN}/nh_font8.c"
    COMMAND "$<TARGET_FILE:nh_hex2c>"
            "${CMAKE_CURRENT_SOURCE_DIR}/font/unscii8.hex"
            "${GEN}/nh_font8.c" nh_font8
    DEPENDS nh_hex2c "${CMAKE_CURRENT_SOURCE_DIR}/font/unscii8.hex"
    COMMENT "hex2c: embedding unscii-8 font")

# NOTE: each generated file is consumed by exactly one target (the
# blobs by nethack, the font by nh_pdcurses) -- attaching an output to
# two targets duplicates its rule under Makefile generators and races
# parallel builds.

# --- PDCursesMod ------------------------------------------------------
file(GLOB PDC_CORE "${pdcursesmod_SOURCE_DIR}/pdcurses/*.c")
if(NETHACK_UI STREQUAL "sdl3")
    file(GLOB PDC_BACKEND "${CMAKE_CURRENT_SOURCE_DIR}/pdcsdl3/*.c")
else()
    file(GLOB PDC_BACKEND "${pdcursesmod_SOURCE_DIR}/${NETHACK_UI}/*.c")
endif()

add_library(nh_pdcurses STATIC ${PDC_CORE} ${PDC_BACKEND})
# PDC_NCMOUSE selects the ncurses-style mouse API and is strictly a
# consumer-side define; compiling the library itself with it breaks
# pdcurses/mouse.c.
target_compile_definitions(nh_pdcurses PUBLIC PDC_WIDE PDC_RGB
                                       INTERFACE PDC_NCMOUSE)
target_include_directories(nh_pdcurses PUBLIC "${pdcursesmod_SOURCE_DIR}")
if(NETHACK_UI STREQUAL "sdl3")
    target_include_directories(nh_pdcurses PRIVATE
        "${CMAKE_CURRENT_SOURCE_DIR}/pdcsdl3")
    target_sources(nh_pdcurses PRIVATE "${GEN}/nh_font8.c")
    target_link_libraries(nh_pdcurses PUBLIC SDL3::SDL3-static)
endif()

# --- the game ---------------------------------------------------------
file(GLOB NH_CORE_SRC "${NH_ROOT}/src/*.c")
set(NH_CURSES_SRC)
foreach(f cursdial cursinit cursinvt cursmain cursmesg cursmisc cursstat
          curswins)
    list(APPEND NH_CURSES_SRC "${NH_ROOT}/win/curses/${f}.c")
endforeach()

if(WIN32)
    set(NH_REGEX_SRC "${NH_ROOT}/sys/share/cppregex.cpp")
else()
    set(NH_REGEX_SRC "${NH_ROOT}/sys/share/posixregex.c")
endif()

add_executable(nethack
    ${NH_CORE_SRC}
    ${NH_CURSES_SRC}
    ${NH_REGEX_SRC}
    "${CMAKE_CURRENT_SOURCE_DIR}/sdl3main.c"
    "${CMAKE_CURRENT_SOURCE_DIR}/sdl3sys.c"
    "${GEN}/nh_symbols_blob.c"
    "${GEN}/nhdat_blob.c")

target_compile_definitions(nethack PRIVATE
    NETHACK_SDL3 DLB DLBMEM
    CURSES_GRAPHICS NOTTYGRAPHICS NOCRASHREPORT
    CURSES_UNICODE CURSES_DEFINE_ERASE_CHAR
    $<$<BOOL:${NETHACK_SDL3_PORTABLE}>:NETHACK_SDL3_PORTABLE>)
target_include_directories(nethack PRIVATE
    "${GENINC}" "${NH_ROOT}/include" "${CMAKE_CURRENT_SOURCE_DIR}")
if(WIN32)
    # win32api.h and friends
    target_include_directories(nethack PRIVATE "${NH_ROOT}/sys/windows")
endif()
target_link_libraries(nethack PRIVATE nh_pdcurses nh_lua SDL3::SDL3-static)

if(WIN32)
    target_link_libraries(nethack PRIVATE bcrypt)
    target_link_options(nethack PRIVATE -static)
    if(NOT NETHACK_SDL3_CONSOLE)
        set_target_properties(nethack PROPERTIES WIN32_EXECUTABLE TRUE)
    endif()

    # Executable icon: decode the stock uuencoded .ico and compile it in
    # as resource #1 (also used for the SDL window's title bar).
    enable_language(RC)
    add_executable(nh_uudecode "${NH_ROOT}/sys/share/uudecode.c")
    add_custom_command(
        OUTPUT "${GEN}/nethack.ico"
        COMMAND "$<TARGET_FILE:nh_uudecode>" "${NH_ROOT}/sys/windows/nhico.uu"
        WORKING_DIRECTORY "${GEN}"
        DEPENDS nh_uudecode "${NH_ROOT}/sys/windows/nhico.uu"
        COMMENT "uudecode: extracting nethack.ico")
    file(WRITE "${GEN}/nethack.rc" "1 ICON \"nethack.ico\"\n")
    set_property(SOURCE "${GEN}/nethack.rc" PROPERTY
        OBJECT_DEPENDS "${GEN}/nethack.ico")
    target_sources(nethack PRIVATE "${GEN}/nethack.rc")
endif()

# Version stamp: only date.c consumes these.
execute_process(COMMAND git rev-parse HEAD
    WORKING_DIRECTORY "${NH_ROOT}" OUTPUT_VARIABLE NH_GIT_SHA
    OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
execute_process(COMMAND git rev-parse --abbrev-ref HEAD
    WORKING_DIRECTORY "${NH_ROOT}" OUTPUT_VARIABLE NH_GIT_BRANCH
    OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
if(NH_GIT_SHA)
    set_property(SOURCE "${NH_ROOT}/src/date.c" APPEND PROPERTY
        COMPILE_DEFINITIONS
        NETHACK_GIT_SHA="${NH_GIT_SHA}" NETHACK_GIT_BRANCH="${NH_GIT_BRANCH}")
endif()
